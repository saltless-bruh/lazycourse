function Term:hide()
  if self.win and vim.api.nvim_win_is_valid(self.win) then
    -- The last window cannot be closed (E444): show an empty buffer there instead.
    if not pcall(vim.api.nvim_win_close, self.win, true) then
      vim.api.nvim_win_call(self.win, function() vim.cmd("enew") end)
    end
  end
  self.win = nil
end

  if self.job then vim.fn.jobstop(self.job); self.job = nil end

  self.buf = nil
end

/** Atomic install of a skill folder: write into a dot temp sibling, swap the old one aside, then drop it. */
export function installSkillDir(root: string, name: string, write: (dir: string) => void): void {
  const dir = join(root, name);
  const tmp = join(root, `.${name}.tmp-${process.pid}`);
  const old = join(root, `.${name}.old-${process.pid}`);
  rmSync(tmp, { recursive: true, force: true });
  mkdirSync(tmp, { recursive: true });
  try { write(tmp); } catch (e) { rmSync(tmp, { recursive: true, force: true }); throw e; }
  rmSync(old, { recursive: true, force: true });
  if (existsSync(dir)) renameSync(dir, old);
  renameSync(tmp, dir);
  rmSync(old, { recursive: true, force: true });
}
